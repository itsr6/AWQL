// Calculate Chemical Oxygen Demand (COD) from BOD and temperature
float calculateCOD(float bod, float temperature) {
  return 0.1011 + (6.7610 * bod) - (0.0204 * temperature);
}

// Calculate BOD values - MODIFIED for 10-minute mark calculation
void calculateBOD() {
  if (!qualityData.isValid || qualityData.do_level <= 0) return;
  
  // Record first measurement time if this is the first valid reading
  if (qualityData.firstMeasurementTime == 0) {
    qualityData.firstMeasurementTime = millis();
    Serial.println("First DO measurement recorded. Will calculate BOD/COD after 10 minutes.");
    return;
  }
  
  // Check if 10 minutes have passed since first measurement
  if ((millis() - qualityData.firstMeasurementTime) < 600000) { // 10 minutes = 600000 ms
    unsigned long remainingTime = (600000 - (millis() - qualityData.firstMeasurementTime)) / 1000;
    Serial.println("Waiting for BOD/COD calculation. Time remaining: " + String(remainingTime) + "s");
    return;
  }
  
  lastBODCalculationTime = millis();
  float currentDO = qualityData.do_level;
  float temp = qualityData.temperature > 0 ? qualityData.temperature : 25.0;
  
  // Calculate DO saturation based on temperature
  qualityData.do_saturation = 14.259 * exp(-0.022 * temp);
  
  // Use default k value if needed
  qualityData.k = 0.4; // Fixed k value
  
  // Calculate BOD values using k and DO deficit
  float doDeficit = qualityData.do_saturation - currentDO;
  if (doDeficit < 0) doDeficit = 0; // No deficit if DO exceeds saturation
  
  // BOD calculation: BOD = DO deficit / k
  qualityData.bod1 = doDeficit / qualityData.k;
  qualityData.bod5 = qualityData.bod1 * 1.5; // Typical BOD5/BOD1 ratio
  
  // Calculate COD values using the specified formula
  qualityData.cod = calculateCOD(qualityData.bod5, temp);
  
  // Constrain to reasonable values
  qualityData.bod1 = constrain(qualityData.bod1, 0, 9999);
  qualityData.bod5 = constrain(qualityData.bod5, 0, 9999);
  qualityData.cod = constrain(qualityData.cod, 0, 9999);
  
  bodCalculated = true;
  
  Serial.println("BOD and COD Calculated:");
  Serial.print("  Temperature: "); 
  Serial.print(temp, 1); 
  Serial.println("°C");
  
  Serial.print("  DO Measured: "); 
  Serial.print(currentDO, 2); 
  Serial.println(" mg/L");
  
  Serial.print("  DO Saturation: "); 
  Serial.print(qualityData.do_saturation, 2); 
  Serial.println(" mg/L");
  
  Serial.print("  BOD₁: "); 
  Serial.print(qualityData.bod1, 2); 
  Serial.println(" mg/L");
  
  Serial.print("  BOD₅: "); 
  Serial.print(qualityData.bod5, 2); 
  Serial.println(" mg/L");
  
  Serial.print("  COD₅: "); 
  Serial.print(qualityData.cod, 2); 
  Serial.println(" mg/L");
}

// Function to initialize NTP time sync
void initializeTime() {
  if (wifiInitialized) {
    Serial.println("Configuring time via NTP...");
    
    // Configure NTP servers
    configTime(gmtOffset_sec, daylightOffset_sec, ntpServer1, ntpServer2);
    
    // Wait for time to be set
    time_t now = 0;
    struct tm timeinfo;
    int retry = 0;
    const int maxRetry = 10;
    
    while (time(&now) < 1000000000 && retry < maxRetry) {
      Serial.print(".");
      delay(500);
      retry++;
    }
    
    if (retry < maxRetry) {
      timeInitialized = true;
      getLocalTime(&timeinfo);
      Serial.println("\nTime initialized successfully!");
      Serial.print("Current time: ");
      Serial.println(&timeinfo, "%Y-%m-%d %H:%M:%S");
      
      // Display time sync status
      tft.fillRect(0, tft.height()-35, tft.width(), 15, TFT_BLACK);
      tft.setCursor(5, tft.height()-30);
      tft.setTextColor(VALUE_COLOR);
      tft.setTextSize(1);
      tft.print("Time sync: ");
      tft.print(&timeinfo, "%Y-%m-%d %H:%M:%S");
    } else {
      Serial.println("\nFailed to initialize time via NTP");
      
      // Display time sync failure
      tft.fillRect(0, tft.height()-35, tft.width(), 15, TFT_BLACK);
      tft.setCursor(5, tft.height()-30);
      tft.setTextColor(ERROR_COLOR);
      tft.setTextSize(1);
      tft.print("Time sync failed!");
    }
    
    lastTimeSync = millis();
  } else {
    Serial.println("Cannot initialize time: WiFi not connected");
  }
}

// Function to get formatted timestamp string (YYYY-MM-DD_HH-MM-SS)
bool getFormattedTimestamp(char* buffer, size_t bufferSize) {
  struct tm timeinfo;
  if (!getLocalTime(&timeinfo)) {
    strcpy(buffer, "0000-00-00_00-00-00");
    return false;
  }
  
  strftime(buffer, bufferSize, "%Y-%m-%d_%H-%M-%S", &timeinfo);
  return true;
}

// Function to log system events to Firebase
void logSystemEvent(const String& eventType, const String& message) {
  if (!firebaseInitialized) return;
  
  // Get current timestamp
  getFormattedTimestamp(timeStampBuf, sizeof(timeStampBuf));
  String timestamp = String(timeStampBuf);
  
  FirebaseJson json;
  json.set("timestamp", timestamp);
  json.set("event", eventType);
  json.set("message", message);
  
  // Create a unique path for this event
  String eventPath = "system/logs/" + timestamp + "_" + String(random(0, 1000));
  
  if (Firebase.RTDB.setJSON(&fbdo, eventPath, &json)) {
    if (DEBUG_FIREBASE) {
      Serial.println("System event logged: " + eventType);
    }
  } else {
    Serial.println("Failed to log system event: " + fbdo.errorReason());
  }
}

// Upload data to Firebase with simplified structure
void uploadToFirebase() {
  if (!firebaseInitialized) {
    Serial.println("Cannot upload: Firebase not initialized");
    return;
  }
  
  if (DEBUG_FIREBASE) {
    Serial.println("Uploading data to Firebase...");
  }
  
  // Create timestamp string
  getFormattedTimestamp(timeStampBuf, sizeof(timeStampBuf));
  String timestamp = String(timeStampBuf);
  
  if (timestamp == "0000-00-00_00-00-00") {
    Serial.println("Warning: Using fallback timestamp (time not synchronized)");
    // Use a fallback timestamp based on device uptime
    timestamp = String(millis()); 
  }
  
  // Only create a record if at least one of the data sets is valid
  if (levelData.isValid || qualityData.isValid) {
    // Create the path with the timestamp
    // Format: water-quality/user/record/YYYY-MM-DD_HH-MM-SS
    String recordPath = "water-quality/user/record/" + timestamp;
    
    FirebaseJson json;
    json.set("timestamp", timestamp);
    
    // Add water level data if valid
    if (levelData.isValid) {
      json.set("depth", levelData.depth);
      json.set("turbidity", levelData.turbidity);
      
      // Add RGB data as a combined string and individual values
      if (levelData.rgbR > 0 || levelData.rgbG > 0 || levelData.rgbB > 0) {
        json.set("rgb", "R" + String(levelData.rgbR) + " G" + String(levelData.rgbG) + 
                        " B" + String(levelData.rgbB) + " C" + String(levelData.rgbC));
        
        // Calculate lux from RGB-C sensor (approximate calculation)
        float luxValue = (levelData.rgbR + levelData.rgbG + levelData.rgbB) * 0.25; 
        json.set("rgb_lux", luxValue);
      }
    }
    
    // Add water quality data if valid
    if (qualityData.isValid) {
      json.set("ph", qualityData.pH);
      json.set("ec", qualityData.ec);
      json.set("do", qualityData.do_level);
      json.set("temp", qualityData.temperature);
      json.set("tds", qualityData.tds);
      
      // Add DO saturation to the database
      // If it's not calculated yet, calculate it here
      float doSaturation;
      if (bodCalculated) {
        // Use the already calculated value if BOD has been calculated
        doSaturation = qualityData.do_saturation;
      } else {
        // Calculate DO saturation based on temperature
        // Formula: DO saturation (mg/L) = 14.59 * e^(-0.0219 * temp)
        float temp = qualityData.temperature > 0 ? qualityData.temperature : 25.0;
        doSaturation = 14.259 * exp(-0.022 * temp);
      }
      json.set("do_saturation", doSaturation);
      
      // Always include BOD/COD values - they'll be 0 until calculated
      json.set("bod1", bodCalculated ? qualityData.bod1 : 0);
      json.set("bod5", bodCalculated ? qualityData.bod5 : 0);
      json.set("cod", bodCalculated ? qualityData.cod : 0);
    }
    
    // Store the record
    if (Firebase.RTDB.setJSON(&fbdo, recordPath, &json)) {
      if (DEBUG_FIREBASE) {
        Serial.println("Water quality record saved at: " + recordPath);
      }
      firebaseUploads++;
    } else {
      Serial.println("Error saving water quality record: " + fbdo.errorReason());
      firebaseErrors++;
    }
  }
  
  lastFirebaseUpload = millis();
}

// Function to retrieve the most recent record
bool getLatestRecord() {
  if (!firebaseInitialized) {
    Serial.println("Cannot retrieve latest record: Firebase not initialized");
    return false;
  }
  
  // Create a QueryFilter object first
  QueryFilter query;
  query.orderBy("$key");
  query.limitToLast(1);
  
  // Then pass a pointer to the query object
  if (Firebase.RTDB.getJSON(&fbdo, "water-quality/user/record", &query)) {
    Serial.println("Latest record retrieved:");
    Serial.println(fbdo.jsonString());
    return true;
  } else {
    Serial.println("Failed to retrieve latest record: " + fbdo.errorReason());
    return false;
  }
  
  // This should never execute but keeps the compiler happy
  return false;
}

// Function to check WiFi and Firebase connectivity
// Returns true if reconnected successfully
bool checkAndReconnect() {
  bool wasReconnected = false;
  
  // Check WiFi connection
  if (WiFi.status() != WL_CONNECTED) {
    Serial.println("WiFi connection lost. Attempting to reconnect...");
    
    WiFi.reconnect();
    
    // Wait for reconnection with timeout
    unsigned long startAttempt = millis();
    while (WiFi.status() != WL_CONNECTED && millis() - startAttempt < 10000) {
      delay(500);
      Serial.print(".");
    }
    
    if (WiFi.status() == WL_CONNECTED) {
      Serial.println("\nWiFi reconnected!");
      Serial.print("IP address: ");
      Serial.println(WiFi.localIP());
      wifiInitialized = true;
      wasReconnected = true;
      
      // Sync time again after reconnection
      initializeTime();
    } else {
      Serial.println("\nWiFi reconnection failed!");
      wifiInitialized = false;
    }
  }
  
  // Check Firebase connection if WiFi is connected
  if (wifiInitialized && !Firebase.ready()) {
    Serial.println("Firebase connection lost. Attempting to reconnect...");
    
    // Initialize Firebase again
    initializeFirebase();
    
    if (firebaseInitialized) {
      Serial.println("Firebase reconnected successfully!");
      wasReconnected = true;
      
      // Log reconnection event
      logSystemEvent("reconnect", "Firebase reconnected after connection loss");
    }
  }
  
  return wasReconnected;
}

// CRC-16 calculation for packet integrity
uint16_t calculateCRC16(uint8_t *data, size_t length) {
  uint16_t crc = 0xFFFF;
  
  if (DEBUG_CRC) {
    Serial.print("CRC Calculation for ");
    Serial.print(length);
    Serial.println(" bytes:");
    
    for (size_t i = 0; i < length; i++) {
      Serial.print("Byte ");
      Serial.print(i);
      Serial.print(": 0x");
      Serial.print(data[i], HEX);
      Serial.print(" -> ");
      
      crc ^= (uint16_t)data[i];
      
      for (uint8_t j = 0; j < 8; j++) {
        if (crc & 0x0001) {
          crc = (crc >> 1) ^ 0xA001;
        } else {
          crc = crc >> 1;
        }
      }
      
      Serial.print("CRC: 0x");
      Serial.println(crc, HEX);
    }
    
    Serial.print("Final CRC: 0x");
    Serial.println(crc, HEX);
  } else {
    for (size_t i = 0; i < length; i++) {
      crc ^= (uint16_t)data[i];
      
      for (uint8_t j = 0; j < 8; j++) {
        crc = (crc & 0x0001) ? ((crc >> 1) ^ 0xA001) : (crc >> 1);
      }
    }
  }
  
  return crc;
}

// Print packet in hexadecimal for debugging
void printPacketHex(uint8_t *packet, int length) {
  Serial.println("Packet data (hex):");
  for (int i = 0; i < length; i++) {
    if (packet[i] < 0x10) Serial.print("0");
    Serial.print(packet[i], HEX);
    Serial.print(" ");
    if ((i + 1) % 8 == 0) Serial.println();
  }
  Serial.println();
}

// Process water level binary packet - UPDATED FOR 16-BYTE FORMAT WITH RGB
void processLevelPacket(int packetSize, int rssi) {
  Serial.print("Processing Node 1 level packet, ");
  Serial.print(packetSize);
  Serial.println(" bytes");
  
  // New packet size is 16 bytes - updated from the original 12 bytes
  if (packetSize != 16) {
    Serial.println("Error: Invalid packet size for water level data (" + String(packetSize) + " bytes)");
    return;
  }
  
  // Read packet
  uint8_t packet[16];
  for (int i = 0; i < packetSize; i++) {
    packet[i] = LoRa.read();
  }
  
  if (DEBUG_PACKET_DATA) {
    printPacketHex(packet, packetSize);
  }
  
  // Verify CRC
  uint16_t receivedCRC = (packet[13] << 8) | packet[14];
  uint16_t calculatedCRC = calculateCRC16(packet, 13);
  
  if (receivedCRC != calculatedCRC) {
    Serial.println("Error: CRC check failed for water level packet");
    Serial.print("Received: 0x");
    Serial.print(receivedCRC, HEX);
    Serial.print(", Calculated: 0x");
    Serial.println(calculatedCRC, HEX);
    crcErrorsNode1++;
    return;
  }
  
  // Extract data
  uint8_t deviceId = packet[0];
  
  // Depth (2 bytes)
  float depth = ((packet[1] << 8) | packet[2]) / 10.0;
  
  // Clear value from RGB sensor (2 bytes)
  uint16_t clearValue = (packet[3] << 8) | packet[4];
  
  // Turbidity (2 bytes)
  float turbidity = ((packet[5] << 8) | packet[6]) / 10.0;
  
  // Turbidity voltage (2 bytes)
  float turbVoltage = ((packet[7] << 8) | packet[8]) / 1000.0;
  
  // RGB values (3 bytes, scaled from 0-255 to 0-65535)
  uint16_t rgbR = packet[9] * 257;  // Scale 8-bit (0-255) to 16-bit (0-65535)
  uint16_t rgbG = packet[10] * 257;
  uint16_t rgbB = packet[11] * 257;
  
  // Sequence number (1 byte)
  uint8_t sequence = packet[12];
  
  // Update current data
  levelData.deviceId = deviceId;
  levelData.depth = depth;
  levelData.turbidity = turbidity;
  levelData.turbVoltage = turbVoltage;
  levelData.rgbR = rgbR;
  levelData.rgbG = rgbG;
  levelData.rgbB = rgbB;
  levelData.rgbC = clearValue;
  levelData.rssi = rssi;
  levelData.timestamp = millis();
  levelData.sequence = sequence;
  levelData.isValid = true;
  
  // Check for missed packets
  static uint8_t lastSequence = 255;
  if (lastSequence != 255 && sequence != ((lastSequence + 1) % 256)) {
    Serial.println("Warning: Missed " + String(((sequence - lastSequence - 1 + 256) % 256)) + " packets from Node 1");
  }
  lastSequence = sequence;
  
  lastLevelDataTime = millis();
  receivedPacketsNode1++;
  receivedPacketsTotal++;
  
  // Log data
  Serial.println("Node 1 water level data processed successfully:");
  Serial.println("  Depth: " + String(levelData.depth) + " cm");
  Serial.println("  Turbidity: " + String(levelData.turbidity) + " NTU");
  Serial.println("  RGB: R=" + String(levelData.rgbR) + 
                 ", G=" + String(levelData.rgbG) + 
                 ", B=" + String(levelData.rgbB) + 
                 ", C=" + String(levelData.rgbC));
  Serial.println("  RSSI: " + String(levelData.rssi) + " dBm");
  Serial.println("  Sequence: " + String(levelData.sequence));
}

// Process water quality binary packet with raw voltage values
void processQualityPacket(int packetSize, int rssi) {
  Serial.print("Processing Node 2 quality packet, ");
  Serial.print(packetSize);
  Serial.println(" bytes");
  
  // Updated water quality packet format (18 bytes total)
  if (packetSize != 18) {
    Serial.println("Error: Invalid packet size for water quality data (" + String(packetSize) + " bytes)");
    return;
  }
  
  // Read packet
  uint8_t packet[18];
  for (int i = 0; i < packetSize; i++) {
    packet[i] = LoRa.read();
  }
  
  if (DEBUG_PACKET_DATA) {
    printPacketHex(packet, packetSize);
  }
  
  // Verify CRC
  uint16_t receivedCRC = (packet[16] << 8) | packet[17];
  uint16_t calculatedCRC = calculateCRC16(packet, 16);
  
  if (receivedCRC != calculatedCRC) {
    Serial.println("Error: CRC check failed for water quality packet");
    Serial.print("Received: 0x");
    Serial.print(receivedCRC, HEX);
    Serial.print(", Calculated: 0x");
    Serial.println(calculatedCRC, HEX);
    crcErrorsNode2++;
    return;
  }
  
  // Extract data
  uint8_t deviceId = packet[0];
  
  // pH (2 bytes) - divided by 100 to get pH value with 2 decimal places
  float pH = ((packet[1] << 8) | packet[2]) / 100.0;
  
  // EC (2 bytes) - divided by 100 to get mS/cm with 2 decimal places
  float ec = ((packet[3] << 8) | packet[4]) / 100.0 * 1000.0; // Convert to µS/cm
  
  // DO (2 bytes) - divided by 100 to get mg/L with 2 decimal places
  float doValue = ((packet[5] << 8) | packet[6]) / 100.0;
  
  // TDS (2 bytes) - raw ppm value
  float tds = ((packet[7] << 8) | packet[8]);
  
 // Temperature (2 bytes) - divided by 100 to get °C with 2 decimal places
 float temperature = ((int16_t)(packet[9] << 8) | packet[10]) / 100.0;
  
 // Sequence number (1 byte)
 uint8_t sequence = packet[11];
 
 // Status/flags (1 byte)
 uint8_t flags = packet[12];
 
 // Reserved bytes (3 bytes)
 // packet[13], packet[14], packet[15]
 
 // Direct values are available, so no need for voltage conversion
 qualityData.pH = pH;
 qualityData.ec = ec;
 qualityData.do_level = doValue;
 qualityData.tds = tds;
 qualityData.temperature = temperature;
 
 // Update metadata
 qualityData.deviceId = deviceId;
 qualityData.rssi = rssi;
 qualityData.timestamp = millis();
 qualityData.sequence = sequence;
 qualityData.isValid = true;
 
 // Check for missed packets
 static uint8_t lastQualitySequence = 255;
 if (lastQualitySequence != 255 && sequence != ((lastQualitySequence + 1) % 256)) {
   Serial.println("Warning: Missed " + String(((sequence - lastQualitySequence - 1 + 256) % 256)) + " packets from Node 2");
 }
 lastQualitySequence = sequence;
 
 lastQualityDataTime = millis();
 receivedPacketsNode2++;
 receivedPacketsTotal++;
 
 // Calculate BOD and COD if needed
 if (qualityData.firstMeasurementTime == 0) {
   // First measurement - record time
   qualityData.firstMeasurementTime = millis();
   Serial.println("First DO measurement recorded at " + String(millis() / 1000) + "s. Will calculate BOD/COD after 10 minutes.");
 } else if ((millis() - qualityData.firstMeasurementTime) >= 600000 && 
            (millis() - lastBODCalculationTime > BOD_UPDATE_INTERVAL)) {
   // If 10 minutes have passed since first measurement and update interval reached
   calculateBOD();
 }
 
 // Log data
 Serial.println("Node 2 water quality data processed successfully:");
 Serial.println("  pH: " + String(qualityData.pH, 2));
 Serial.println("  EC: " + String(qualityData.ec, 2) + " uS/cm");
 Serial.println("  DO: " + String(qualityData.do_level, 2) + " mg/L");
 Serial.println("  TDS: " + String(qualityData.tds, 0) + " ppm");
 Serial.println("  Temperature: " + String(qualityData.temperature, 2) + " °C");
 Serial.println("  RSSI: " + String(qualityData.rssi) + " dBm");
 Serial.println("  Sequence: " + String(qualityData.sequence));
}

// Process JSON packet
void processJsonPacket(String message, int rssi) {
 Serial.println("Processing JSON packet: " + message);

 JsonDocument doc;
 DeserializationError error = deserializeJson(doc, message);

 if (error) {
   Serial.println("JSON parsing failed: " + String(error.c_str()));
   return;
 }

 // Check packet type by fields
 if (doc.containsKey("depth")) {
   // Water level packet (Node 1) - with RGB sensor
   levelData.depth = doc["depth"].as<float>();
   levelData.turbidity = doc["turb_ntu"].as<float>();
   levelData.turbVoltage = doc["turb_v"].as<float>();
   levelData.deviceId = doc["id"] | 1;  // Default to 1 if not present
   
   // Check for RGB values
   if (doc.containsKey("rgb_r")) {
     levelData.rgbR = doc["rgb_r"].as<uint16_t>();
     levelData.rgbG = doc["rgb_g"].as<uint16_t>();
     levelData.rgbB = doc["rgb_b"].as<uint16_t>();
     
     if (doc.containsKey("rgb_c")) {
       levelData.rgbC = doc["rgb_c"].as<uint16_t>();
     }
   }
   
   levelData.rssi = rssi;
   levelData.timestamp = millis();
   levelData.sequence++;
   levelData.isValid = true;
   
   lastLevelDataTime = millis();
   receivedPacketsNode1++;
   receivedPacketsTotal++;
 } 
 else if (doc.containsKey("ph") || doc.containsKey("phVolt")) {
   // Water quality packet (Node 2)
   // Check if we have direct values or voltages
   if (doc.containsKey("ph")) {
     // Direct values provided
     qualityData.pH = doc["ph"].as<float>();
     qualityData.ec = doc["ec"].as<float>() * 1000.0; // Convert mS/cm to µS/cm
     qualityData.do_level = doc["do"].as<float>();
     qualityData.tds = doc["tds"].as<float>();
     
     // Get temperature if available
     if (doc.containsKey("temp")) {
       qualityData.temperature = doc["temp"].as<float>();
     }
   }
   
   qualityData.deviceId = doc["id"] | 2;  // Default to 2 if not present
   qualityData.rssi = rssi;
   qualityData.timestamp = millis();
   qualityData.sequence++;
   qualityData.isValid = true;
   
   lastQualityDataTime = millis();
   receivedPacketsNode2++;
   receivedPacketsTotal++;
   
   // Calculate BOD and COD if needed
   if (qualityData.firstMeasurementTime == 0) {
     // First measurement - record time
     qualityData.firstMeasurementTime = millis();
     Serial.println("First DO measurement recorded at " + String(millis() / 1000) + "s. Will calculate BOD/COD after 10 minutes.");
   } else if ((millis() - qualityData.firstMeasurementTime) >= 600000 && 
              (millis() - lastBODCalculationTime > BOD_UPDATE_INTERVAL)) {
     // If 10 minutes have passed since first measurement and update interval reached
     calculateBOD();
   }
 }
}

// Check for packets with timeout
void checkForPacketsWithTimeout() {
 unsigned long startCheck = millis();
 const unsigned long MAX_CHECK_TIME = 500;
 
 int packetSize = LoRa.parsePacket();
 while (packetSize == 0 && (millis() - startCheck < MAX_CHECK_TIME)) {
   packetSize = LoRa.parsePacket();
   delay(1);
 }
 
 if (packetSize > 0) {
   int rssi = LoRa.packetRssi();
   
   Serial.println("\n=== RECEIVED LORA PACKET ===");
   Serial.println("Size: " + String(packetSize) + " bytes");
   Serial.println("RSSI: " + String(rssi) + " dBm");
   
   // Process by packet size
   if (packetSize == 16) {
     // Water level packet (Node 1 with RGB sensors)
     processLevelPacket(packetSize, rssi);
     flashReceivedIndicator(tft.width()-10, 10, TFT_GREEN);
   } 
   else if (packetSize == 18) {
     // Water quality packet (Node 2)
     processQualityPacket(packetSize, rssi);
     flashReceivedIndicator(tft.width()-20, 10, RGB_COLOR);
   }
   else {
     // Try to process as JSON
     String message = "";
     while (LoRa.available()) {
       message += (char)LoRa.read();
     }
     processJsonPacket(message, rssi);
     flashReceivedIndicator(tft.width()-30, 10, SYNC_COLOR);
   }
   
   // Update display with new data
   updateDisplay();
   
   // Always put LoRa back into receive mode
   LoRa.receive();
 }
}

// Configure LoRa parameters
void configureLoRa() {
 LoRa.setSpreadingFactor(8);
 LoRa.setSignalBandwidth(250E3);
 LoRa.setCodingRate4(6);
 LoRa.setSyncWord(0xF1);
 LoRa.enableCrc();
 LoRa.receive();
 
 if (DEBUG_LORA) {
   Serial.println("LoRa configured with parameters:");
   Serial.println("- Spreading Factor: 8");
   Serial.println("- Bandwidth: 250 kHz");
   Serial.println("- Coding Rate: 4/6");
   Serial.println("- Sync Word: 0xF1");
   Serial.println("- CRC: Enabled");
 }
}

// Check LoRa status and reinitialize if needed
void checkLoRaStatus() {
 if (!loraInitialized && (millis() - lastReceiveAttempt > RECEIVE_RETRY_INTERVAL)) {
   Serial.println("Attempting to initialize LoRa...");
   
   // Reset LoRa module
   pinMode(LORA_RST, OUTPUT);
   digitalWrite(LORA_RST, LOW);
   delay(20);
   digitalWrite(LORA_RST, HIGH);
   delay(50);
   
   // Initialize SPI for LoRa
   LoRaSPI.begin(LORA_SCK, LORA_MISO, LORA_MOSI, LORA_SS);
   LoRa.setSPI(LoRaSPI);
   LoRa.setPins(LORA_SS, LORA_RST, LORA_DIO0);
   
   // Try with timeout
   unsigned long startTime = millis();
   bool success = false;
   
   while ((millis() - startTime) < 3000) {
     if (LoRa.begin(915E6)) {
       success = true;
       break;
     }
     delay(100);
   }
   
   if (success) {
     loraInitialized = true;
     configureLoRa();
     updateDisplay();
   }
   
   lastReceiveAttempt = millis();
 } 
 // Reset receive mode if no data for a while
 else if (loraInitialized && 
         (millis() - lastLevelDataTime > 30000) && 
         (millis() - lastQualityDataTime > 30000) &&  
         (millis() - lastReceiveAttempt > RECEIVE_RETRY_INTERVAL)) {
   if (DEBUG_LORA) {
     Serial.println("No data received for 30+ seconds, resetting LoRa receive mode");
   }
   LoRa.receive();
   lastReceiveAttempt = millis();
 }
}

// Check WiFi status and retry connection if needed
void checkWiFiStatus() {
 if (!wifiInitialized && (millis() - lastWiFiCheck > 60000)) {  // Try every minute
   Serial.println("Attempting to reconnect WiFi...");
   initializeWiFi();
   
   if (wifiInitialized && !firebaseInitialized) {
     initializeFirebase();
   }
   
   lastWiFiCheck = millis();
 }
}

// Main setup function
void setup() {
 // Initialize serial
 Serial.begin(115200);
 delay(1000);
 
 Serial.println("\n*** Water Quality Monitoring System - Receiver ***");
 Serial.println("ESP32-S3 Optimized - Separate SPI Bus Version");
 Serial.println("Firebase Integration with NTP Time Sync");
 Serial.println("Build: " __DATE__ " " __TIME__);
 
 // Record startup time
 startupTime = millis();
 
 // Initialize display
 initializeDisplay();
 initializeDisplayLayout();
 
 // Initialize LoRa
 Serial.println("Initializing LoRa module...");
 
 // Reset LoRa module
 pinMode(LORA_RST, OUTPUT);
 digitalWrite(LORA_RST, LOW);
 delay(20);
 digitalWrite(LORA_RST, HIGH);
 delay(50);
 
 // Initialize SPI for LoRa
 LoRaSPI.begin(LORA_SCK, LORA_MISO, LORA_MOSI, LORA_SS);
 LoRa.setSPI(LoRaSPI);
 LoRa.setPins(LORA_SS, LORA_RST, LORA_DIO0);
 
 // Try to initialize with timeout
 bool success = false;
 unsigned long startAttempt = millis();
 
 Serial.print("Attempting LoRa.begin(915E6)... ");
 
 // Try for 5 seconds
 while (millis() - startAttempt < 5000) {
   if (LoRa.begin(915E6)) {
     success = true;
     break;
   }
   Serial.print(".");
   delay(200);
 }
 
 if (success) {
   loraInitialized = true;
   Serial.println("\nLoRa initialized successfully!");
   configureLoRa();
 } else {
   Serial.println("\nLoRa initialization timed out. Will retry later.");
 }
 
 lastReceiveAttempt = millis();
 
 // Initialize WiFi
 initializeWiFi();
 
 // Initialize time if WiFi is connected
 if (wifiInitialized) {
   initializeTime();
 }
 
 // Initialize Firebase if WiFi is connected
 if (wifiInitialized) {
   initializeFirebase();
 }
 
 // Show initial screen
 updateDisplay();
 
 Serial.println("\n=== SYSTEM READY ===");
}

// Main loop function
void loop() {
 // Print debug message every 30 seconds
 static unsigned long lastDebugMsg = 0;
 if (millis() - lastDebugMsg > DEBUG_MSG_INTERVAL) {
   Serial.println("\n----- System Status -----");
   Serial.println("Time elapsed: " + String((millis() - startupTime) / 1000) + "s");
   Serial.println("LoRa Status: " + String(loraInitialized ? "ONLINE" : "OFFLINE"));
   Serial.println("WiFi Status: " + String(wifiInitialized ? "CONN=" : "DIS"));
   Serial.println("Firebase Status: " + String(firebaseInitialized ? "CON" : "DIS"));
   
   // Print timestamp if time is synchronized
   if (timeInitialized) {
     getFormattedTimestamp(timeStampBuf, sizeof(timeStampBuf));
     Serial.println("Current time: " + String(timeStampBuf));
   }
   
   Serial.println("Packets: Total=" + String(receivedPacketsTotal) + 
                  ", Node1=" + String(receivedPacketsNode1) + 
                  ", Node2=" + String(receivedPacketsNode2));
   Serial.println("CRC Errors: Node1=" + String(crcErrorsNode1) + 
                  ", Node2=" + String(crcErrorsNode2));
   Serial.println("Firebase: Uploads=" + String(firebaseUploads) + 
                  ", Errors=" + String(firebaseErrors));
   
   // Force LoRa back to receive mode
   if (loraInitialized) {
     LoRa.receive();
     Serial.println("LoRa receive mode refreshed");
   }
   
   lastDebugMsg = millis();
 }
 
 // Check LoRa status and try to reinitialize if needed
 checkLoRaStatus();
 
 // Check WiFi status and try to reconnect if needed
 checkWiFiStatus();
 
 // Check for connection issues and try to reconnect
 checkAndReconnect();
 
 // Check for packets with timeout
 checkForPacketsWithTimeout();
 
 // Calculate BOD periodically if we have quality data
 if (qualityData.isValid && qualityData.firstMeasurementTime > 0 && 
     (millis() - qualityData.firstMeasurementTime) >= 600000 && 
     (millis() - lastBODCalculationTime > BOD_UPDATE_INTERVAL)) {
   calculateBOD();
 }
 
 // Upload data to Firebase periodically
 if (firebaseInitialized && (millis() - lastFirebaseUpload > FIREBASE_UPLOAD_INTERVAL)) {
   if (levelData.isValid || qualityData.isValid) {
     uploadToFirebase();
   }
 }
 
 // Check if time needs to be resynced (every 24 hours)
 if (wifiInitialized && (millis() - lastTimeSync > 86400000)) { // 24 hours
   initializeTime();
 }
 
 // Update display periodically
 static unsigned long lastDisplayUpdate = 0;
 if (millis() - lastDisplayUpdate > DISPLAY_UPDATE_INTERVAL) {
   updateDisplay();
   lastDisplayUpdate = millis();
 }
 
 // delay to prevent CPU overload
 delay(10);
}
